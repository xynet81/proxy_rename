# proxy_test_exit_ip.py 功能分析与 JavaScript 实现计划

## 1. 功能分析

### 1.1 核心功能

* **代理节点处理**：支持 VMess、VLESS、SS、Trojan、Hysteria2 等多种协议
* **订阅获取**：从多个订阅 URL 获取并解析代理节点
* **端口过滤**：自动过滤使用端口 53 的节点
* **Xray 配置生成**：将代理节点转换为 Xray 配置
* **出口 IP 测试**：通过代理发送请求，获取出口 IP 及地理信息
* **Webhook API**：提供 HTTP 端点处理订阅请求和输出结果
* **定时任务**：定期从订阅 URL 获取节点并更新结果

### 1.2 技术栈

* **Python 3.12+**
* **Flask**：构建 Webhook API
* **asyncio**：异步编程
* **httpx**：异步 HTTP 客户端
* **subprocess**：调用外部 Xray 程序
* **schedule**：定时任务
* **logging**：日志记录
* **tempfile**：临时文件管理

#### 3.3.1 订阅获取和解析

```javascript
const axios = require('axios');
const base64 = require('base64-js');

async function fetchSubscriptions(subUrls) {
    const results = await Promise.all(subUrls.map(fetchSingleSubscription));
    const allUris = new Set();
    results.forEach(uris => uris.forEach(uri => allUris.add(uri)));
    return Array.from(allUris);
}

async function fetchSingleSubscription(url) {
    try {
        const response = await axios.get(url, { timeout: 10000 });
        let content = response.data.trim();
        
        // 尝试 base64 解码
        try {
            content = Buffer.from(content, 'base64').toString('utf-8');
        } catch (error) {
            // 非 base64，直接使用
        }
        
        const lines = content.split('\n');
        return lines.filter(line => line.trim() && isValidUri(line.trim()));
    } catch (error) {
        console.error(`Failed to fetch subscription from ${url}:`, error.message);
        return [];
    }
}
```

#### 3.3.3 出口 IP 测试

```javascript
const { spawn } = require('child_process');
const axios = require('axios');
const SocksProxyAgent = require('socks-proxy-agent');

async function testExitIp(configInfo) {
    const { config, port, uri } = configInfo;
    
    // 写入临时配置文件
    const configFile = await writeTempConfig(config);
    
    // 启动 Xray 进程
    const proc = spawn('xray', ['run', '-c', configFile], {
        stdio: 'ignore',
        detached: false
    });
    
    // 等待 Xray 启动
    await new Promise(resolve => setTimeout(resolve, 2000));
    
    try {
        // 通过 SOCKS 代理发送请求
        const agent = new SocksProxyAgent(`socks5://127.0.0.1:${port}`);
        const response = await axios.get('http://ip-api.com/json?fields=status,query', {
            httpAgent: agent,
            httpsAgent: agent,
            timeout: 10000
        });
        
        // 解析响应
        const data = response.data;
        if (data.status === 'success') {
            const exitIp = data.query;
            // 获取地理信息
            const geoInfo = await getGeoInfo(exitIp, agent);
            return {
                exitIp,
                countryCode: geoInfo.countryCode || '未知',
                region: geoInfo.region || '未知',
                status: 'success',
                reachable: true
            };
        }
        return {
            exitIp: '未知',
            countryCode: '未知',
            region: '未知',
            status: 'API response failed',
            reachable: false
        };
    } catch (error) {
        return {
            exitIp: '未知',
            countryCode: '未知',
            region: '未知',
            status: `Test failed: ${error.message}`,
            reachable: false
        };
    } finally {
        // 终止 Xray 进程
        proc.kill();
        // 删除临时文件
        await deleteTempConfig(configFile);
    }
}
```

#### 3.3.4 Webhook API

```javascript
const express = require('express');
const app = express();
const bodyParser = require('body-parser');

app.use(bodyParser.json());

// API 密钥验证中间件
const apiKeyMiddleware = (req, res, next) => {
    const apiKey = req.headers['x-api-key'];
    if (apiKey !== process.env.API_KEY) {
        return res.status(401).json({ error: 'Invalid API Key' });
    }
    next();
};

// 处理订阅请求
app.post('/process', apiKeyMiddleware, async (req, res) => {
    try {
        const { sub_urls, max_workers = 4, include_details = false } = req.body;
        
        // 获取订阅节点
        const uris = await fetchSubscriptions(sub_urls);
        
        // 处理节点
        const results = await processBatch(uris, max_workers);
        
        // 生成响应
        const response = {
            new_sub_b64: generateNewSubscription(results),
            success_count: results.filter(r => r.reachable).length,
            total_nodes: results.length,
            success_rate: results.length > 0 ? 
                Math.round((results.filter(r => r.reachable).length / results.length) * 100 * 100) / 100 : 0
        };
        
        if (include_details) {
            response.results = results;
        }
        
        res.json(response);
    } catch (error) {
        res.status(500).json({ error: error.message });
    }
});

// 订阅输出端点

        status: 'healthy',

    });
```
